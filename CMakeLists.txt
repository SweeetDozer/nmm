cmake_minimum_required(VERSION 3.22)
project(MusicOrder VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent Multimedia Network)
find_package(TagLib 2.0 CONFIG REQUIRED)
add_library(musiccore src/core.cpp)
target_include_directories(musiccore PUBLIC src)
target_link_libraries(musiccore PUBLIC Qt6::Core TagLib::TagLib)
add_library(musicui src/window.cpp src/search.cpp src/http.cpp src/online.cpp src/online_dialogs.cpp src/acoustid.cpp src/discogs.cpp)
target_link_libraries(musicui PUBLIC musiccore Qt6::Widgets Qt6::Concurrent Qt6::Multimedia Qt6::Network)
add_executable(musicorder src/main.cpp)
target_link_libraries(musicorder PRIVATE musicui)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(catalog_test tests/catalog_test.cpp)
 target_link_libraries(catalog_test PRIVATE musicui Qt6::Test)
 add_test(NAME catalog_search COMMAND catalog_test)
 set_tests_properties(catalog_search PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
 add_executable(online_test tests/online_test.cpp)
 target_link_libraries(online_test PRIVATE musicui Qt6::Test)
 add_test(NAME online_choices COMMAND online_test)
 set_tests_properties(online_choices PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
 add_executable(search_test tests/search_test.cpp)
 target_link_libraries(search_test PRIVATE musicui Qt6::Test)
 add_test(NAME search_evidence COMMAND search_test)
 add_executable(ui_test tests/ui_test.cpp)
 target_link_libraries(ui_test PRIVATE musicui Qt6::Test)
 add_test(NAME collection_5000 COMMAND ui_test)
 set_tests_properties(collection_5000 PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
 add_executable(core_test tests/core_test.cpp)
 target_link_libraries(core_test PRIVATE musiccore Qt6::Gui)
 find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
 find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
 add_test(NAME tag_roundtrip COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_roundtrip.py $<TARGET_FILE:core_test>)
 add_test(NAME ui_smoke COMMAND musicorder --smoke-test)
 set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS musicorder RUNTIME DESTINATION bin)
